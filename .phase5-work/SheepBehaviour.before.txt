using UnityEngine;

namespace FarmingSimulation
{
    [RequireComponent(typeof(CharacterController), typeof(AnimalStats))]
    public sealed class SheepBehaviour : MonoBehaviour
    {
        [SerializeField] private AnimalEnclosure enclosure;
        [SerializeField] private Animator animator;
        [SerializeField, Min(0.1f)] private float moveSpeed = 0.7f;
        [SerializeField, Min(1f)] private float turnSpeed = 120f;
        [SerializeField, Min(0.1f)] private float idleDuration = 3f;
        [SerializeField, Min(0.1f)] private float eatingDuration = 4f;
        [SerializeField, Min(0)] private float feedingCooldown = 10f;
        [SerializeField, Range(0, 100)] private float hungryThreshold = 65f;
        [SerializeField, Range(0, 100)] private float minimumHungerToFeed = 25f;
        [SerializeField, Min(1)] private float foodAmount = 35f;
        [SerializeField] private int randomSeed = 123;
        private CharacterController controller;
        private AnimalStats stats;
        private System.Random random;
        private Vector3 destination;
        private float stateTime, nextFeedTime, verticalSpeed;
        private bool cooldownWasActive;
        public AnimalStats Stats => stats;
        public bool CoolingDown => Time.time < nextFeedTime;
        public bool NeedsFood => stats.Hunger >= minimumHungerToFeed;
        public event System.Action FeedbackChanged;
        private static readonly int Fed = Animator.StringToHash("Fed");

        private void Awake()
        {
            controller = GetComponent<CharacterController>();
            stats = GetComponent<AnimalStats>();
            random = new System.Random(randomSeed);
            if (enclosure == null || animator == null)
            {
                Debug.LogError("SheepBehaviour needs its enclosure and Animator assigned.", this);
                enabled = false;
                return;
            }
            animator.applyRootMotion = false;
            SetState(AnimalBehaviour.Idle, idleDuration);
        }

        private void SetState(AnimalBehaviour state, float seconds)
        {
            stateTime = seconds;
            stats.SetBehaviour(state);
            FeedbackChanged?.Invoke();
        }

        private void Update()
        {
            if (cooldownWasActive && !CoolingDown)
            {
                cooldownWasActive = false;
                FeedbackChanged?.Invoke();
            }
            stateTime -= Time.deltaTime;
            Vector3 velocity = Vector3.zero;
            if (stats.Behaviour == AnimalBehaviour.Eating)
            {
                if (stateTime <= 0f) SetState(AnimalBehaviour.Idle, idleDuration);
            }
            else if (stats.Hunger >= hungryThreshold)
            {
                if (stats.Behaviour != AnimalBehaviour.Hungry) SetState(AnimalBehaviour.Hungry, 0f);
            }
            else if (stats.Behaviour == AnimalBehaviour.Hungry)
                SetState(AnimalBehaviour.Idle, idleDuration);
            else if (stats.Behaviour == AnimalBehaviour.Idle && stateTime <= 0f)
                ChooseDestination();
            else if (stats.Behaviour == AnimalBehaviour.Wander)
            {
                Vector3 offset = destination - transform.position;
                offset.y = 0f;
                if (offset.magnitude < 0.15f || stateTime <= 0f
                    || !PathIsClear(transform.position + offset.normalized * Mathf.Min(0.8f, offset.magnitude)))
                    SetState(AnimalBehaviour.Idle, idleDuration);
                else
                {
                    velocity = offset.normalized * Mathf.Min(moveSpeed, offset.magnitude / Mathf.Max(Time.deltaTime, 0.001f));
                    transform.rotation = Quaternion.RotateTowards(transform.rotation,
                        Quaternion.LookRotation(offset), turnSpeed * Time.deltaTime);
                }
            }

            Vector3 next = enclosure.ClampPosition(transform.position + velocity * Time.deltaTime);
            Vector3 step = next - transform.position;
            step.y = 0f;
            if (controller.isGrounded && verticalSpeed < 0f) verticalSpeed = -2f;
            verticalSpeed = Mathf.Max(verticalSpeed - 20f * Time.deltaTime, -30f);
            CollisionFlags contacts = controller.Move(step + Vector3.up * verticalSpeed * Time.deltaTime);
            if ((contacts & CollisionFlags.Sides) != 0 && stats.Behaviour == AnimalBehaviour.Wander)
                SetState(AnimalBehaviour.Idle, idleDuration);
        }

        private void ChooseDestination()
        {
            for (int attempt = 0; attempt < 12; attempt++)
            {
                Vector3 candidate = enclosure.SampleDestination(random, transform.position.y);
                if (Vector3.Distance(candidate, transform.position) > 0.8f && PathIsClear(candidate))
                {
                    destination = candidate;
                    SetState(AnimalBehaviour.Wander, 15f);
                    return;
                }
            }
            SetState(AnimalBehaviour.Idle, idleDuration);
        }

        private bool PathIsClear(Vector3 point)
        {
            Vector3 offset = point - transform.position;
            offset.y = 0f;
            // Sphere stays above the floor. Full path check rejects destinations behind obstacles.
            float clearance = controller.radius + 0.05f;
            return !PlayerPhysicsQueries.SphereCast(transform.position + Vector3.up * (clearance + 0.1f),
                clearance, offset.normalized, offset.magnitude, transform, out _);
        }

        public bool TryFeed()
        {
            if (!isActiveAndEnabled || !NeedsFood || CoolingDown || stats.Behaviour == AnimalBehaviour.Eating)
                return false;
            nextFeedTime = Time.time + feedingCooldown;
            cooldownWasActive = true;
            stats.ReduceHunger(foodAmount);
            SetState(AnimalBehaviour.Eating, eatingDuration);
            // One happy jump per accepted feeding; Idle remains the movement/eating placeholder.
            animator.SetTrigger(Fed);
            return true;
        }
    }
}
